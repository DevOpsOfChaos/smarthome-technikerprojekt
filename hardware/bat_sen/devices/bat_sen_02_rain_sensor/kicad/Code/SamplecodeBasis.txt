#include <WiFi.h>
#include <esp_now.h>

// MAC-Adresse des Masters eintragen (als Array)
uint8_t masterAddress[] = {0x24, 0x6F, 0x28, 0xAA, 0xBB, 0xCC}; // Beispiel: anpassen!

// Struktur für Sensordaten
typedef struct struct_message {
  float temperature;
  int battery;
} struct_message;

struct_message dataToSend;

// Callback: Sende-Status ausgeben
void OnDataSent(const uint8_t *mac_addr, esp_now_send_status_t status) {
  Serial.print("Send Status: ");
  Serial.println(status == ESP_NOW_SEND_SUCCESS ? "Success" : "Fail");
}

void setup() {
  Serial.begin(115200);

  WiFi.mode(WIFI_STA); // Station-Modus
  WiFi.disconnect();

  if (esp_now_init() != ESP_OK) {
    Serial.println("Error initializing ESP-NOW");
    return;
  }

  esp_now_peer_info_t peerInfo = {};
  memcpy(peerInfo.peer_addr, masterAddress, 6);
  peerInfo.channel = 6; // Kanal ggf. anpassen!
  peerInfo.encrypt = false;

  if (esp_now_add_peer(&peerInfo) != ESP_OK) {
    Serial.println("Failed to add peer");
    return;
  }

  esp_now_register_send_cb(OnDataSent);

  // Beispiel-Daten (hier: feste Werte, sensorspezifisch anpassen)
  dataToSend.temperature = 25.7;
  dataToSend.battery = 97;

  // Daten senden
  esp_err_t result = esp_now_send(masterAddress, (uint8_t *)&dataToSend, sizeof(dataToSend));
  if (result == ESP_OK) {
    Serial.println("Data sent successfully");
  } else {
    Serial.println("Sending error");
  }

  delay(100); // warten bis Sendung abgeschlossen ist

  // DeepSleep für 10 Minuten
  esp_sleep_enable_timer_wakeup(10 * 60 * 1000000);
  Serial.println("Entering deep sleep now...");
  esp_deep_sleep_start();
}

void loop() {
  // nicht genutzt: nach deep sleep wird nur setup() ausgeführt
}
